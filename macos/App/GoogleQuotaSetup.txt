from pathlib import Path
from threading import Thread
from time import sleep
from http.server import ThreadingHTTPServer
from ai_usage import antigravity_quota, store
from ai_usage.server import Dashboard

class GoogleQuota(Dashboard):
    db_path = Path.cwd().parent / "quota.sqlite"

    def send(self, status, body, *args, **kwargs):
        if isinstance(body, dict) and "plans" in body and "totals" in body:
            body = dict(body, has_history=False, plans=[p for p in body["plans"] if p["vendor"] == "google"])
        super().send(status, body, *args, **kwargs)

def refresh():
    db = store.connect(GoogleQuota.db_path, init=False)
    while True:
        try:
            antigravity_quota.poll(db)
        except Exception:
            print("Quota could not be read. Check that Antigravity is open and signed in.", flush=True)
        sleep(300)

try:
    listener = ThreadingHTTPServer(("127.0.0.1", 8796), GoogleQuota)
except OSError:
    raise SystemExit("Port 8796 is in use. If your AI Usage dashboard is already running, connect to it in the app.")
store.connect(GoogleQuota.db_path).close()
Thread(target=refresh, daemon=True).start()
print("Google quota reader ready at localhost:8796. Leave this window open. Control-C stops it.", flush=True)
with listener:
    try:
        listener.serve_forever()
    except KeyboardInterrupt:
        pass
PYTHON
